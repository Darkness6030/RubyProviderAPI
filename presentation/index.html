05
Ruby-сервис

Сервис с подтверждённым контрактом

class Provider
  class NovapayService < BaseService
    BASE_URL = ENV.fetch("NOVAPAY_BASE_URL", ...)

    def create_request(
      operation, request_method = nil)
      path = expand_path("/payouts", operation)
      response = client.public_send("post",
        "#{BASE_URL}#{path}",
        **request_options(operation, request_method))
      handle_create_response(response)
    end

    def fetch_status(operation)
      path = expand_path("/payouts/{payout_id}", operation)
      response = client.public_send("get",
        "#{BASE_URL}#{path}", **status_request_options(operation))
      handle_status_response(response)
    end
  end
end
